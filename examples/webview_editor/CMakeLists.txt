cmake_minimum_required(VERSION 3.24)
project(CellWebEditorExample LANGUAGES CXX)
if(NOT TARGET Cell::WebView)
    find_package(Cell CONFIG REQUIRED COMPONENTS Core DSP WebView)
endif()
add_library(cell_web_example_processor STATIC ../sdk_gain/processor.cpp)
target_include_directories(cell_web_example_processor PUBLIC ../sdk_gain)
target_link_libraries(cell_web_example_processor PUBLIC Cell::DSP)
add_executable(cell_web_bridge_example bridge.cpp)
target_link_libraries(cell_web_bridge_example PRIVATE cell_web_example_processor Cell::WebView)
option(CELL_WEB_EXAMPLE_WINDOW "Build the optional native-window browser demo" OFF)
if(CELL_WEB_EXAMPLE_WINDOW)
    if(NOT TARGET Cell::UI)
        find_package(Cell CONFIG REQUIRED COMPONENTS UI WebView)
    endif()
    add_executable(cell_web_editor_example main.cpp)
    target_link_libraries(cell_web_editor_example PRIVATE cell_web_example_processor Cell::WebView Cell::UI)
endif()
