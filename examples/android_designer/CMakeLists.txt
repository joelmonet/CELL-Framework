cmake_minimum_required(VERSION 3.24)
project(CellAndroidDesigner LANGUAGES CXX)

if(NOT ANDROID)
    message(FATAL_ERROR "Configure the Designer app with the Android NDK toolchain")
endif()
set(CELL_ANDROID_SDK_ROOT "" CACHE PATH "Directory containing one installed Cell SDK per ABI")
if(CELL_ANDROID_SDK_ROOT AND NOT Cell_DIR)
    set(Cell_DIR "${CELL_ANDROID_SDK_ROOT}/${ANDROID_ABI}/lib/cmake/Cell")
endif()
find_package(Cell CONFIG REQUIRED COMPONENTS DSP AudioDevice)

set(CELL_ANDROID_EDITOR_DIR "" CACHE PATH "Optional directory with both Cell UI Designer exports")
if(CELL_ANDROID_EDITOR_DIR)
    set(html "${CELL_ANDROID_EDITOR_DIR}/cell-plugin-editor.html")
    set(bindings "${CELL_ANDROID_EDITOR_DIR}/cell-ui-bindings.hpp")
else()
    set(html "${CMAKE_CURRENT_SOURCE_DIR}/../auv3_gain/Cell Gain Editor.html")
    set(bindings "${CMAKE_CURRENT_SOURCE_DIR}/../auv3_gain/cell-ui-bindings.hpp")
endif()
add_library(cell_android_gain STATIC "../sdk_gain/processor.cpp")
set_target_properties(cell_android_gain PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_link_libraries(cell_android_gain PRIVATE Cell::DSP)
cell_add_android_designer_app(cell_android_designer cell_android_gain "${html}" "${bindings}")
