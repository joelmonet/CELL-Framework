cmake_minimum_required(VERSION 3.24)
project(CellTransportConsumer VERSION 0.1.0 LANGUAGES CXX)
find_package(Cell CONFIG REQUIRED COMPONENTS Core Wrappers)
include(CTest)
add_library(cell_transport_fixture STATIC processor.cpp)
set_target_properties(cell_transport_fixture PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_link_libraries(cell_transport_fixture PUBLIC Cell::Core)
if(CMAKE_SYSTEM_NAME STREQUAL "Darwin")
    enable_language(OBJCXX)
    get_property(adapters GLOBAL PROPERTY CELL_ADAPTER_SOURCE_DIR)
    foreach(kind IN ITEMS host midi_fx_host instrument_host)
        set(target cell_transport_auv3_${kind})
        add_executable(${target} auv3_${kind}.mm "${adapters}/auv3.mm")
        set_target_properties(${target} PROPERTIES
            OBJCXX_STANDARD 20 OBJCXX_STANDARD_REQUIRED ON OBJCXX_EXTENSIONS OFF)
        target_compile_options(${target} PRIVATE -fobjc-arc)
        target_link_libraries(${target} PRIVATE Cell::Core
            "-framework Foundation" "-framework AudioToolbox" "-framework AVFoundation" "-framework CoreMIDI")
    endforeach()
    set(auv3_example "${CMAKE_CURRENT_SOURCE_DIR}/../../examples/auv3_gain")
    target_include_directories(cell_transport_auv3_midi_fx_host PRIVATE "${auv3_example}")
    target_sources(cell_transport_auv3_midi_fx_host PRIVATE
        "${auv3_example}/midi_processor.cpp" "${auv3_example}/midi_check.mm")
    target_include_directories(cell_transport_auv3_instrument_host PRIVATE "${auv3_example}")
    target_sources(cell_transport_auv3_instrument_host PRIVATE
        "${auv3_example}/instrument_processor.cpp" "${auv3_example}/instrument_check.mm")
    add_test(NAME transport_auv3 COMMAND cell_transport_auv3_host)
    add_test(NAME transport_auv3_ump COMMAND cell_transport_auv3_host --ump)
    add_test(NAME midi_effect_auv3 COMMAND cell_transport_auv3_midi_fx_host)
    add_test(NAME instrument_auv3 COMMAND cell_transport_auv3_instrument_host)
endif()
set(CELL_TRANSPORT_CLAP_SDK "" CACHE PATH "Separately supplied official CLAP checkout")
set(CELL_TRANSPORT_VST3_SDK "" CACHE PATH "Separately supplied official VST3 checkout")
if(BUILD_TESTING AND (CELL_TRANSPORT_CLAP_SDK OR CELL_TRANSPORT_VST3_SDK))
    set(transport_dump "${CMAKE_CURRENT_BINARY_DIR}/transport-test.syx")
    string(ASCII 240 125 1 2 247 240 125 3 4 5 247 transport_dump_bytes)
    file(WRITE "${transport_dump}" "${transport_dump_bytes}")
endif()
if(CELL_TRANSPORT_CLAP_SDK)
    cell_add_clap_plugin(CellTransport_CLAP PROCESSOR cell_transport_fixture
        ID org.cell.test.transport NAME "Cell Transport Test" CLAP_SDK "${CELL_TRANSPORT_CLAP_SDK}" CLAP_EXPOSE_PROGRAMS)
    add_executable(cell_transport_clap_host host.cpp)
    target_compile_definitions(cell_transport_clap_host PRIVATE CELL_TRANSPORT_CLAP=1)
    target_include_directories(cell_transport_clap_host PRIVATE "${CELL_TRANSPORT_CLAP_SDK}/include")
    target_link_libraries(cell_transport_clap_host PRIVATE Cell::Core ${CMAKE_DL_LIBS})
    target_compile_features(cell_transport_clap_host PRIVATE cxx_std_20)
    add_test(NAME transport_clap COMMAND cell_transport_clap_host "$<TARGET_FILE:CellTransport_CLAP>")
    if(BUILD_TESTING)
        add_test(NAME transport_clap_dump COMMAND cell_transport_clap_host "$<TARGET_FILE:CellTransport_CLAP>"
            --dump "${transport_dump}")
    endif()
endif()
if(CELL_TRANSPORT_VST3_SDK)
    foreach(option IN ITEMS SMTG_ENABLE_VST3_PLUGIN_EXAMPLES SMTG_ENABLE_VST3_HOSTING_EXAMPLES SMTG_ENABLE_VSTGUI_SUPPORT SMTG_CREATE_PLUGIN_LINK SMTG_RUN_VST_VALIDATOR SMTG_ENABLE_WAYLAND_SUPPORT)
        set(${option} OFF CACHE BOOL "" FORCE)
    endforeach()
    add_subdirectory("${CELL_TRANSPORT_VST3_SDK}" "${CMAKE_CURRENT_BINARY_DIR}/vst3sdk" EXCLUDE_FROM_ALL)
    cell_add_vst3_plugin(CellTransport_VST3 PROCESSOR cell_transport_fixture
        ID org.cell.test.transport NAME "Cell Transport Test" COMPANY "Cell" CATEGORIES "Fx|Tools")
    if(APPLE)
        enable_language(OBJCXX)
        set(module "${CELL_TRANSPORT_VST3_SDK}/public.sdk/source/vst/hosting/module_mac.mm")
        set_source_files_properties("${module}" PROPERTIES COMPILE_FLAGS "-fobjc-arc")
    elseif(WIN32)
        set(module "${CELL_TRANSPORT_VST3_SDK}/public.sdk/source/vst/hosting/module_win32.cpp")
    else()
        set(module "${CELL_TRANSPORT_VST3_SDK}/public.sdk/source/vst/hosting/module_linux.cpp")
    endif()
    add_executable(cell_transport_vst3_host host.cpp "${module}"
        "${CELL_TRANSPORT_VST3_SDK}/public.sdk/source/vst/hosting/plugprovider.cpp")
    target_compile_features(cell_transport_vst3_host PRIVATE cxx_std_20)
    target_link_libraries(cell_transport_vst3_host PRIVATE Cell::Core sdk_hosting ${CMAKE_DL_LIBS})
    if(APPLE)
        target_link_libraries(cell_transport_vst3_host PRIVATE "-framework Cocoa")
    endif()
    get_target_property(bundle CellTransport_VST3 SMTG_PLUGIN_PACKAGE_PATH)
    add_test(NAME transport_vst3 COMMAND cell_transport_vst3_host "${bundle}")
    if(BUILD_TESTING)
        add_test(NAME transport_vst3_dump COMMAND cell_transport_vst3_host "${bundle}" --dump "${transport_dump}")
    endif()
endif()

    set_tests_properties(transport_clap_editor PROPERTIES TIMEOUT 45 SKIP_RETURN_CODE 77)
endif()
